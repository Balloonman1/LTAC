function status(r){
  if(!r.has_ltac_grants) return {cls:'off',label:'No LTAC grant pool',key:'none'};
  if(r.cycle==='rolling') return {cls:'ok',label:'Rolling',key:'open'};
  if(!r.deadline) return {cls:'warn',label:'Date not posted',key:'unknown'};
  const n = daysUntil(r.deadline);
  if(n>=0 && n<=14) return {cls:'bad',label:n===0?'Due today':n===1?'Due tomorrow':`Due in ${n} days`,key:'open'};
  if(n>14) return {cls:'ok',label:'Open',key:'open'};
  return {cls:'off',label:'Closed',key:'closed'};
}
function nextExpected(r){
  if(!r.has_ltac_grants || !r.deadline || r.cycle==='rolling') return null;
  if(daysUntil(r.deadline)>=0) return null;
  let d=r.deadline, step = r.cycle==='biennial'?2:1;
  while(daysUntil(d)<0) d=addYear(d,step);
  return d;
}

function matches(r){
  if(mineF && !follows.has(r.name)) return false;
  if(typeF!=='all' && r.type!==typeF) return false;
  if(countyF && r.county!==countyF) return false;
  if(regionF && r.region!==regionF) return false;
  if(cycF!=='all' && r.cycle!==cycF) return false;
  const s=status(r);
  if(statF==='open' && s.key!=='open') return false;
  if(statF==='pres' && !(r.presentation&&(r.presentation.required||r.presentation.date))) return false;
  if(statF==='fp' && !(r.for_profit==='yes'||r.for_profit==='sponsor')) return false;
  if(q){
    const hay=[r.name,r.county,r.notes,r.contact,r.deadline_pattern,...(r.members||[]).flatMap(m=>[m.name,m.affiliation,m.role]),...(r.alternatives||[]).flatMap(a=>[a.program,a.administrator]),...(r.funding_docs||[]).map(d=>d.title)].join(' ').toLowerCase();
    if(!hay.includes(q)) return false;
  }
  return true;
}
function sortKey(r){
  if(sortF==='pool') return [0,-(r.competitive_pool_usd??-1),r.name];
  if(sortF==='total') return [0,-(r.total_pool_usd??-1),r.name];
  if(sortF==='name') return [0,0,r.name];
  const s=status(r);
  const rank={open:0,unknown:1,closed:2,none:3}[s.key];
  const d = s.key==='open'&&r.deadline?daysUntil(r.deadline):(nextExpected(r)?daysUntil(nextExpected(r)):9999);
  return [rank,d,r.name];
}

const API='/api/store';
async function apiFetch(q,opts){const r=await fetch(API+q,Object.assign({headers:{'content-type':'application/json'}},opts||{}));const d=await r.json().catch(()=>({}));if(!r.ok)throw Object.assign(new Error(d.error||r.statusText),{code:r.status});return d;}
const netlifyDb={
  doc(path){ const email=decodeURIComponent(path.replace(/^lists\//,'')); return {
    async get(){ const d=await apiFetch('?c=lists&email='+encodeURIComponent(email)); return {exists:d.exists,data:()=>d.data}; },
    async set(v){ await apiFetch('?c=lists',{method:'POST',body:JSON.stringify({email,follows:v.follows||[],touch:true})}); },
    async update(v){ const b={email}; if(v.follows) b.follows=v.follows; if('visits' in v) b.touch=true; await apiFetch('?c=lists',{method:'POST',body:JSON.stringify(b)}); }
  }; },
  collection(c){ return {
    async add(rec){ await apiFetch('?c='+c,{method:'POST',body:JSON.stringify(rec)}); },
    orderBy(){return this;}, limit(){return this;},
    async get(){ const d=await apiFetch('?c='+c); return {docs:(d.items||[]).map(x=>({data:()=>x}))}; }
  }; }
};

async function signIn(email){
  const ref=db.doc('lists/'+email.replace(/[^A-Za-z0-9_\-.~:@+]/g,'_'));
  let snap=null; try{snap=await ref.get();}catch(e){}
  const now=new Date().toISOString();
  let uid=null; try{uid=userCap?await userCap.id():null;}catch(e){}
  if(snap&&snap.exists){
    const d=snap.data(); follows=new Set(Array.isArray(d.follows)?d.follows:[]);
    try{await ref.update({lastSeen:now,visits:(d.visits||0)+1,userId:uid||d.userId||null});}catch(e){}
  } else {
    follows=new Set();
    try{await ref.set({email,follows:[],created:now,lastSeen:now,visits:1,userId:uid});}
    catch(e){if(e&&e.code==='invalid_argument'){account=null;viewOnly();return;}}
  }
  account={email,ref}; try{localStorage.setItem('ltac-account',email);}catch(e){}
  if(pendingFollow){ follows.add(pendingFollow); pendingFollow=null; persist(); }
  renderAcct(); render();
}
